""" # Example usage: error_type_name = make_error_type("Invalid Input", error_code=101, error_type="InvalidData") error_instance = create_error_instance(error_type_name, "Input is not valid.") print(f"Created Error Type: {error_type_name}") print(f"Error Instance: {error_instance}") # Example of creating multiple error types error_type_1 = make_error_type("Incorrect Calculation", error_code=102, error_type="CalculationError") error_type_2 = make_error_type("Data Error", error_code=103, error_type="DataError") print("\nMultiple Error Types Created:") print(f"Error Type 1: {error_type_1}") print(f"Error Type 2: {error_type_2}") #Demonstrate how to access the error type print("\nAccessing Error Type:") print(f"Error Type 1: {error_type_1}") print(f"Error Type 2: {error_type_2}") if __name__ == "__main__": main() ```
